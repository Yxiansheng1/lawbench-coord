function Invoke-IsolatedSupervisor([string]$SpecPath) {
  $spec = Get-Content -Raw -LiteralPath $SpecPath | ConvertFrom-Json -AsHashtable
  $stdoutLog = [string]$spec.stdoutLog
  $stderrLog = [string]$spec.stderrLog
  $handoffPath = [string]$spec.handoffPath
  foreach ($logPath in @($stdoutLog, $stderrLog, $handoffPath)) {
    $parent = Split-Path -Parent $logPath
    if ($parent -and -not (Test-Path -LiteralPath $parent -PathType Container)) {
      New-Item -ItemType Directory -Path $parent | Out-Null
    }
  }

  $process = $null
  $stdoutWriter = $null
  $stderrWriter = $null
  try {
    $startInfo = [System.Diagnostics.ProcessStartInfo]::new([string]$spec.processPath)
    $startInfo.UseShellExecute = $false
    $startInfo.WorkingDirectory = [string]$spec.workingDirectory
    $startInfo.RedirectStandardOutput = $true
    $startInfo.RedirectStandardError = $true
    $startInfo.CreateNoWindow = [bool]$spec.createNoWindow
    $startInfo.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Normal
    foreach ($argument in @($spec.arguments)) {
      if ($null -ne $argument) { $startInfo.ArgumentList.Add([string]$argument) }
    }

    $startInfo.Environment.Clear()
    foreach ($entry in $spec.environment.GetEnumerator()) {
      $startInfo.Environment[[string]$entry.Key] = [string]$entry.Value
    }

    $stdoutStream = [System.IO.File]::Open(
      $stdoutLog,
      [System.IO.FileMode]::Append,
      [System.IO.FileAccess]::Write,
      [System.IO.FileShare]::ReadWrite
    )
    $stderrStream = [System.IO.File]::Open(
      $stderrLog,
      [System.IO.FileMode]::Append,
      [System.IO.FileAccess]::Write,
      [System.IO.FileShare]::ReadWrite
    )
    $utf8 = [System.Text.UTF8Encoding]::new($false)
    $stdoutWriter = [System.IO.StreamWriter]::new($stdoutStream, $utf8)
    $stderrWriter = [System.IO.StreamWriter]::new($stderrStream, $utf8)
    $stdoutWriter.AutoFlush = $true
    $stderrWriter.AutoFlush = $true

    $process = [System.Diagnostics.Process]::new()
    $process.StartInfo = $startInfo
    if (-not $process.Start()) { throw 'Process.Start returned false' }
    $process.Refresh()
    $record = [ordered]@{
      ok = $true
      pid = $process.Id
      supervisorPid = $PID
      startTimeUtc = $process.StartTime.ToUniversalTime().ToString('o')
      startTimeUtcTicks = [string]$process.StartTime.ToUniversalTime().Ticks
      executablePath = [System.IO.Path]::GetFullPath([string]$spec.processPath)
    }
    $temporaryHandoff = "$handoffPath.tmp"
    $record | ConvertTo-Json -Depth 4 | Set-Content -LiteralPath $temporaryHandoff -Encoding utf8NoBOM
    Move-Item -Force -LiteralPath $temporaryHandoff -Destination $handoffPath

    $stdoutTask = $process.StandardOutput.ReadLineAsync()
    $stderrTask = $process.StandardError.ReadLineAsync()
    while ($true) {
      $activity = $false
      if ($null -ne $stdoutTask -and $stdoutTask.IsCompleted) {
        $line = $stdoutTask.GetAwaiter().GetResult()
        if ($null -eq $line) {
          $stdoutTask = $null
        } else {
          $stdoutWriter.WriteLine($line)
          $stdoutWriter.Flush()
          $stdoutTask = $process.StandardOutput.ReadLineAsync()
        }
        $activity = $true
      }
      if ($null -ne $stderrTask -and $stderrTask.IsCompleted) {
        $line = $stderrTask.GetAwaiter().GetResult()
        if ($null -eq $line) {
          $stderrTask = $null
        } else {
          $stderrWriter.WriteLine($line)
          $stderrWriter.Flush()
          $stderrTask = $process.StandardError.ReadLineAsync()
        }
        $activity = $true
      }
      if ($process.HasExited -and $null -eq $stdoutTask -and $null -eq $stderrTask) { break }
      if (-not $activity) { Start-Sleep -Milliseconds 20 }
    }
    $process.WaitForExit()
    try {
      $completion = [ordered]@{
        pid = $process.Id
        exitCode = $process.ExitCode
        exitedAtUtc = (Get-Date).ToUniversalTime().ToString('o')
      }
      $completion | ConvertTo-Json -Depth 3 | Set-Content -LiteralPath "$handoffPath.completed.json" -Encoding utf8NoBOM
    } catch {
      $completion = [ordered]@{
        pid = $process.Id
        exitCode = 'unavailable'
        error = $_.Exception.Message
        exitedAtUtc = (Get-Date).ToUniversalTime().ToString('o')
      }
      $completion | ConvertTo-Json -Depth 3 | Set-Content -LiteralPath "$handoffPath.completed.json" -Encoding utf8NoBOM
    }
  } catch {
    if ($process -and -not $process.HasExited) {
      try {
        $process.Kill($true)
        $process.WaitForExit(5000) | Out-Null
      } catch {}
    }
    $failure = [ordered]@{
      ok = $false
      supervisorPid = $PID
      error = $_.Exception.Message
    }
    $failure | ConvertTo-Json -Depth 3 | Set-Content -LiteralPath $handoffPath -Encoding utf8NoBOM
    throw
  } finally {
    if ($stdoutWriter) { $stdoutWriter.Dispose() }
    if ($stderrWriter) { $stderrWriter.Dispose() }
    if ($process) { $process.Dispose() }
  }
}

function Assert-File([string]$Path, [string]$Label) {
  if (-not (Test-Path -LiteralPath $Path -PathType Leaf)) { throw "$Label missing: $Path" }
}

function Assert-Directory([string]$Path, [string]$Label) {

function Assert-Junction([string]$Path, [string]$ExpectedTarget, [string]$Label) {
  if (-not (Test-Path -LiteralPath $Path)) { throw "$Label junction missing: $Path" }
  $item = Get-Item -LiteralPath $Path -Force
  if ($item.LinkType -ne 'Junction') { throw "$Label is not a junction: $Path" }
  $targets = @($item.Target | ForEach-Object { [IO.Path]::GetFullPath($_) })
  $expected = [IO.Path]::GetFullPath($ExpectedTarget)
  if (-not ($targets | Where-Object { $_.TrimEnd('\') -ieq $expected.TrimEnd('\') })) {
    throw "$Label junction target mismatch: $Path -> $($targets -join ';')"
  }
}

function New-IsolatedBaseEnvironment {
  [OutputType([System.Collections.Generic.Dictionary[string, string]])]
  $environment = [System.Collections.Generic.Dictionary[string, string]]::new([StringComparer]::OrdinalIgnoreCase)
  foreach ($name in @(
    'SystemRoot', 'WINDIR', 'SystemDrive', 'PATH', 'PATHEXT', 'COMSPEC',
    'PROCESSOR_ARCHITECTURE', 'NUMBER_OF_PROCESSORS', 'OS', 'USERNAME', 'USERDOMAIN', 'LOGONSERVER', 'SESSIONNAME'
  )) {
    $value = [Environment]::GetEnvironmentVariable($name, [EnvironmentVariableTarget]::Process)
    if ($null -ne $value) { $environment[$name] = $value }
  }
  $homePath = [IO.Path]::GetFullPath($UserHome)
  $environment['HOME'] = $homePath
  $environment['USERPROFILE'] = $homePath
  $environment['HOMEDRIVE'] = [IO.Path]::GetPathRoot($homePath).TrimEnd('\')
  $environment['HOMEPATH'] = $homePath.Substring([IO.Path]::GetPathRoot($homePath).Length - 1)
  $environment['APPDATA'] = $Roaming
  $environment['LOCALAPPDATA'] = $Local
  $environment['PROGRAMDATA'] = $ProgramData
  $environment['TEMP'] = $Temp
  $environment['TMP'] = $Temp
  return $environment
}

function Copy-IsolatedEnvironment {
  [OutputType([System.Collections.Generic.Dictionary[string, string]])]
  param(
    [Parameter(Mandatory)][System.Collections.Generic.Dictionary[string, string]]$SourceEnvironment
  )

  }
  return $environment
}

function Start-IsolatedTarget(
  [string]$Name,
  [string]$ProcessPath,
  [string[]]$Arguments,
  [string]$WorkingDirectory,
  [System.Collections.Generic.Dictionary[string, string]]$Environment,
  [bool]$CreateNoWindow
) {
  $specPath = Join-Path $StateDir "$Name-supervisor-$stamp.json"
  $handoffPath = Join-Path $StateDir "$Name-handoff-$stamp.json"
  $stdoutLog = Join-Path $LogDir "$Name-$stamp.stdout.log"
  $stderrLog = Join-Path $LogDir "$Name-$stamp.stderr.log"
  $spec = [ordered]@{
    processPath = $ProcessPath
    arguments = @($Arguments)
    workingDirectory = $WorkingDirectory
    createNoWindow = $CreateNoWindow
    stdoutLog = $stdoutLog
    stderrLog = $stderrLog
    handoffPath = $handoffPath
    environment = $Environment
  }
  $spec | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath $specPath -Encoding utf8NoBOM

  $supervisor = Start-Process -FilePath $PwshPath -ArgumentList @(
    '-NoProfile',
    '-File', $PSCommandPath,
    '-SupervisorSpecPath', $specPath
  ) -WindowStyle Hidden -PassThru

  $deadline = (Get-Date).AddSeconds(10)
  while (-not (Test-Path -LiteralPath $handoffPath -PathType Leaf)) {
    if ($supervisor.HasExited) { throw "$Name supervisor exited before publishing process identity" }
    if ((Get-Date) -gt $deadline) { throw "$Name supervisor did not publish process identity within 10 seconds" }
    Start-Sleep -Milliseconds 100
  }
  $record = Get-Content -Raw -LiteralPath $handoffPath | ConvertFrom-Json
  if (-not $record.ok) { throw "$Name supervisor failed: $($record.error)" }
  return [pscustomobject]@{
    pid = [int]$record.pid
    supervisorPid = [int]$record.supervisorPid
    startTimeUtc = [string]$record.startTimeUtc
    startTimeUtcTicks = [string]$record.startTimeUtcTicks
    executablePath = [string]$record.executablePath
    stdoutLog = $stdoutLog
    stderrLog = $stderrLog
    specPath = $specPath
    handoffPath = $handoffPath
  }
}

function Assert-RecordedProcess([int]$TargetPid, [string]$ExpectedExecutable, [string]$ExpectedTicks, [string]$Label) {
  $process = Get-Process -Id $TargetPid -ErrorAction SilentlyContinue
  if (-not $process) { throw "$Label is not running at PID $TargetPid" }
  if ([IO.Path]::GetFullPath($process.Path) -ine [IO.Path]::GetFullPath($ExpectedExecutable)) {
    throw "$Label executable mismatch at PID $TargetPid"
  }
  $actualTicks = [string]$process.StartTime.ToUniversalTime().Ticks
  if ($actualTicks -ne $ExpectedTicks) { throw "$Label process start time mismatch at PID $TargetPid" }
}
