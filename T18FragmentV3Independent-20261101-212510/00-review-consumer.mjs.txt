0592 export function validateStage2Result(parsed, inventory) {
0593   exactKeys(parsed, ["reviews"], "stage2", "RESPONSE_SCHEMA_INVALID");
0594   if (!Array.isArray(parsed.reviews)) fail("stage2", "RESPONSE_SCHEMA_INVALID");
0595   if (!inventory || !Array.isArray(inventory.targets)) fail("stage2", "TARGET_TABLE_INVALID");
0596   const targetsById = new Map();
0597   for (const target of inventory.targets) {
0598     const kind = targetKind(target.target_id);
0599     if (!kind || kind !== target.kind || targetsById.has(target.target_id)) {
0600       fail("stage2", "TARGET_TABLE_INVALID");
0601     }
0602     targetsById.set(target.target_id, target);
0603   }
0604   const seen = new Set();
0605   const reviews = [];
0606   for (const record of parsed.reviews) {
0607     exactKeys(record, ["target_id", "treatment", "refs", "assessment"],
0608       "stage2", "REVIEW_SCHEMA_INVALID");
0609     if (!nonemptyString(record.target_id)) fail("stage2", "TARGET_ID_INVALID");
0610     const target = targetsById.get(record.target_id);
0611     if (!target) fail("stage2", "UNKNOWN_TARGET_ID");
0612     if (seen.has(record.target_id)) fail("stage2", "DUPLICATE_TARGET_ID");
0613     seen.add(record.target_id);
0614     const allowedTreatments = target.kind === "F" ? FRAGMENT_TREATMENTS : SUPPORT_TREATMENTS;
0615     if (!allowedTreatments.includes(record.treatment)) fail("stage2", "TREATMENT_INVALID");
0616     if (!nonemptyString(record.assessment) || record.assessment.trim() === "" ||
0617         record.assessment.length > ASSESSMENT_MAX_UTF16) {
0618       fail("stage2", "ASSESSMENT_INVALID");
0619     }
0620     if (!Array.isArray(record.refs)) fail("stage2", "REFS_INVALID");
0621     const allowedKinds = allowedRefKinds(target.kind);
0622     const refIds = new Set();
0623     for (const ref of record.refs) {
0624       if (!nonemptyString(ref)) fail("stage2", "REF_INVALID");
0625       const refTarget = targetsById.get(ref);
0626       if (!refTarget) fail("stage2", "UNKNOWN_REF_TARGET_ID");
0627       if (!allowedKinds.has(refTarget.kind)) fail("stage2", "REF_FAMILY_INVALID");
0628       if (refIds.has(ref)) fail("stage2", "REF_DUPLICATE");
0629       refIds.add(ref);
0630     }
0631     if ((record.treatment === "ADDRESSED" || record.treatment === "SUPPORTED") &&
0632         record.refs.length === 0) {
0633       fail("stage2", "POSITIVE_TREATMENT_WITHOUT_REF");
0634     }
0635     reviews.push({
0636       target_id: record.target_id,
0637       treatment: record.treatment,
0638       assessment: record.assessment,
0639       target: { ...target },
0640       refs: record.refs.map((ref) => ({ ...targetsById.get(ref) })),
0641     });
0642   }
0643   if (inventory.targets.some((target) => !seen.has(target.target_id))) {
0644     fail("stage2", "MISSING_TARGET_REVIEW");
0645   }
0646   if (reviews.length !== inventory.targets.length) fail("stage2", "REVIEW_COUNT_MISMATCH");
0647   const order = new Map(inventory.targets.map((target, position) => [target.target_id, position]));
0648   reviews.sort((left, right) => order.get(left.target_id) - order.get(right.target_id));
0649   return { reviews };
0650 }

0775 async function readCredentialFromStdin() {
0776   if (process.stdin.isTTY) fail("credential", "TTY_NOT_ALLOWED");
0777   return new Promise((resolve, reject) => {
0778     let captured = Buffer.alloc(0);
0779     let settled = false;
0780     const cleanup = () => {
0781       process.stdin.off("data", onData);
0782       process.stdin.off("end", onEnd);
0783       process.stdin.off("error", onError);
0784       if (typeof process.stdin.pause === "function") process.stdin.pause();
0785     };
0786     const finish = () => {
0787       if (settled) return;
0788       settled = true;
0789       cleanup();
0790       let line = captured;
0791       if (line.length > 0 && line[line.length - 1] === 13) line = line.subarray(0, -1);
0792       if (line.length === 0 || line.length > 4096 ||
0793           [...line].some((byte) => byte < 33 || byte > 126)) {
0794         reject(new PilotProtocolError("credential", "CREDENTIAL_INVALID"));
0795         return;
0796       }
0797       resolve(line.toString("ascii"));
0798     };
0799     const onData = (chunk) => {
0800       if (settled) return;
0801       const bytes = Buffer.from(chunk);
0802       captured = Buffer.concat([captured, bytes]);
0803       const newline = captured.indexOf(10);
0804       if (newline >= 0) {
0805         captured = captured.subarray(0, newline);
0806         finish();
0807       } else if (captured.length > 4096) {
0808         settled = true;
0809         cleanup();
0810         reject(new PilotProtocolError("credential", "CREDENTIAL_INVALID"));
0811       }
0812     };
0813     const onEnd = () => finish();
0814     const onError = () => {
0815       if (settled) return;
0816       settled = true;
0817       cleanup();
0818       reject(new PilotProtocolError("credential", "STDIN_FAILED"));
0819     };
0820     process.stdin.on("data", onData);
0821     process.stdin.on("end", onEnd);
0822     process.stdin.on("error", onError);
0823     process.stdin.resume();
0824   });
0825 }

0827 export async function performRequest(stage, requestBody, credential, deadline) {
0828   const startedAt = Date.now();
0829   const timeoutMs = requestTimeoutForDeadline(startedAt, deadline);
0830   if (timeoutMs <= 0) fail(stage, "ENTIRE_CALL_CAP_EXCEEDED");
0831   const signal = AbortSignal.timeout(timeoutMs);
0832   let responseStatus = null;
0833   const chunks = [];
0834   let capturedBytes = 0;
0835   let overflow = false;
0836   try {
0837     const response = await fetch(ROUTE, {
0838       method: "POST",
0839       redirect: "error",
0840       headers: { Authorization: buildAuthorizationHeader(credential), "Content-Type": "application/json" },
0841       body: requestBody,
0842       signal,
0843     });
0844     responseStatus = response.status;
0845     if (response.body) {
0846       const reader = response.body.getReader();
0847       let streamFailed = false;
0848       try {
0849         while (true) {
0850           const { done, value } = await reader.read();
0851           if (done) break;
0852           const chunk = Buffer.from(value);
0853           if (capturedBytes + chunk.byteLength > RESPONSE_CAP_BYTES) {
0854             overflow = true;
0855             await reader.cancel();
0856             break;
0857           }
0858           chunks.push(chunk);
0859           capturedBytes += chunk.byteLength;
0860         }
0861       } catch (error) {
0862         streamFailed = true;
0863         throw error;
0864       } finally {
0865         if (streamFailed) {
0866           try {
0867             await reader.cancel();
0868           } catch {
0869             // Cancel errors are swallowed only for a stream that already failed.
0870           }
0871         }
0872         reader.releaseLock();
0873       }
0874     }
0875     return {
0876       status: response.status,
0877       body: Buffer.concat(chunks).toString("utf8"),
0878       capturedBytes,
0879       responseOverflow: overflow,
0880       durationMs: Date.now() - startedAt,
0881       transportError: null,
0882     };
0883   } catch (error) {
0884     const signalTimedOut = signal.aborted === true &&
0885       signal.reason && signal.reason.name === "TimeoutError";
0886     const timedOut = (error && error.name === "TimeoutError") || signalTimedOut;
0887     const wholeDeadlineExpired = Date.now() >= deadline ||
0888       (timedOut && timeoutMs < PER_CALL_TIMEOUT_MS);
0889     return {
0890       status: responseStatus,
0891       body: Buffer.concat(chunks).toString("utf8"),
0892       capturedBytes,
0893       responseOverflow: overflow,
0894       durationMs: Date.now() - startedAt,
0895       transportError: wholeDeadlineExpired
0896         ? "ENTIRE_CALL_CAP_EXCEEDED"
0897         : timedOut ? "TIMEOUT" : "TRANSPORT_FAILURE",
0898     };
0899   }
0900 }
