# Exact immutable base Firewall
class Firewall:
    def __init__(
        self,
        app,
        *,
        root: pathlib.Path,
        case: pathlib.Path,
        case_id: str,
        host_token: str,
    ) -> None:
        self.app = app
        self.root = root
        self.case = case
        self.case_id = case_id
        self.expected_auth = ("Bearer " + host_token).encode()
        self.ledger = root / "requests.ndjson"
        self.lock = threading.RLock()
        self.records: list[dict[str, Any]] = []
        self.case_open_attempts = 0
        self.case_open_ok = 0
        self.read_ok = 0
        self.auth_denials = 0
        self.policy_denials: dict[str, int] = {}
        with self.ledger.open("x", encoding="utf-8"):
            pass

    def _record(
        self, method: str, path: str, status: int | None, disposition: str, started: float
    ) -> None:
        row = {
            "method": method,
            "path": path,
            "status": status,
            "disposition": disposition,
            "elapsed_ms": int((time.monotonic() - started) * 1000),
        }
        with self.lock:
            self.records.append(row)
            with self.ledger.open("a", encoding="utf-8", newline="\n") as stream:
                stream.write(json.dumps(row, ensure_ascii=False) + "\n")

    def _deny(self, code: str) -> None:
        with self.lock:
            self.policy_denials[code] = self.policy_denials.get(code, 0) + 1

    def _authorized(self, scope) -> bool:
        values = [
            value
            for key, value in scope.get("headers", [])
            if key.lower() == b"authorization"
        ]
        if len(values) == 1 and hmac.compare_digest(values[0], self.expected_auth):
            return True
        with self.lock:
            self.auth_denials += 1
        return False

    def _policy(self, method: str, path: str, data: Any, query: bytes) -> str | None:
        if method == "POST" and path == "/api/case/open":
            if query:
                return "CASE_OPEN_QUERY"
            if not (
                isinstance(data, dict)
                and set(data).issubset({"path", "template", "kind"})
                and {"path", "template"}.issubset(data)
                and data.get("template") is None
                and data.get("kind") in {None, "daily"}
                and isinstance(data.get("path"), str)
                and _norm(data["path"]) == _norm(self.case)
            ):
                return "CASE_OPEN_BODY"
            with self.lock:
                if self.case_open_attempts >= 2:
                    return "CASE_OPEN_LIMIT"
                self.case_open_attempts += 1
            return None
        if method != "GET" or path not in READ_ONLY_ROUTES:
            return "ROUTE_OR_METHOD_NOT_ALLOWED"
        if path in NO_QUERY_ROUTES:
            return None if not query else "UNEXPECTED_QUERY"
        if path == "/api/task/current":
            values = _query_values(query, required={"session_id"}, optional=None)
            if values is None or len(values["session_id"]) > 128:
                return "SESSION_ID_QUERY"
            return None
        if path in CASE_QUERY_ROUTES:
            values = _query_values(query, required={"case_id"}, optional=None)
            if values is None:
                return "CASE_QUERY_EXACT"
            if values["case_id"] != self.case_id:
                return "CASE_ID_OUTSIDE_RUN"
        return None

    def _complete(self, method: str, path: str, status: int | None, response: Any) -> None:
        if path == "/health":
            return
        if method == "POST" and path == "/api/case/open":
            value = response.get("value") if isinstance(response, dict) else None
            if (
                status == 200
                and isinstance(response, dict)
                and response.get("ok") is True
                and isinstance(value, dict)
                and value.get("case_id") == self.case_id
                and value.get("created") is False
            ):
                with self.lock:
                    self.case_open_ok += 1
            return
        if status == 200 and isinstance(response, dict) and response.get("ok") is True:
            with self.lock:
                self.read_ok += 1

    async def __call__(self, scope, receive, send) -> None:
        if scope["type"] != "http":
            return
        method, path, started = scope["method"], scope["path"], time.monotonic()
        if not self._authorized(scope):
            self._record(method, path, 401, "auth_not_forwarded", started)
            await _deny_reply(send, 401, "AUTH_REQUIRED")
            return

        body = b""
        data = None
        if method == "POST":
            try:
                body = await _read_body(receive)
            except Deny:
                self._deny("BODY_TOO_LARGE")
                self._record(method, path, 413, "BODY_TOO_LARGE", started)
                await _deny_reply(send, 413, "BODY_TOO_LARGE")
                return
            if body is None:
                self._record(method, path, None, "disconnected", started)
                return
            data = _json(body)
            used = False

            async def replay():
                nonlocal used
                if not used:
                    used = True
                    return {"type": "http.request", "body": body, "more_body": False}
                return await original_receive()

            original_receive = receive
            receive = replay

        query = scope.get("query_string", b"")
        denial = self._policy(method, path, data, query)
        if denial:
            self._deny(denial)
            self._record(method, path, 403, denial, started)
            await _deny_reply(send, 403, denial)
            return

        status: int | None = None
        captured = bytearray()
        overflow = False

        async def observed(message):
            nonlocal status, overflow
            if message["type"] == "http.response.start":
                status = int(message["status"])
            elif message["type"] == "http.response.body":
                chunk = message.get("body", b"")
                room = max(0, MAX_CAPTURE - len(captured))
                if len(chunk) > room:
                    overflow = True
                captured.extend(chunk[:room])
            await send(message)

        try:
            await self.app(scope, receive, observed)
        finally:
            self._record(method, path, status, "native_app", started)
        if overflow:
            self._deny("RESPONSE_CAPTURE_OVERFLOW")
            return
        self._complete(method, path, status, _json(bytes(captured)))

    def evidence(self) -> dict[str, Any]:
        with self.lock:
            return {
                "requests": len(self.records),
                "read_ok": self.read_ok,
                "case_open_attempts": self.case_open_attempts,
                "case_open_ok": self.case_open_ok,
                "auth_denials": self.auth_denials,
                "policy_denials": dict(sorted(self.policy_denials.items())),
                "delete_route_observed": any(
                    "session-delete" in row["path"] for row in self.records
                ),
            }

 * 在 DSH 原生 beginSubmission 生成 R 后、prompt 开始前调用。
 * 成功后才建立 reservation；prompt 被拒时 release 同一 R，并把界面批次留给律师重试。
 */
export function makeMaterialSubmissionHook(
  cwdOf: (sessionId: string) => string | undefined,
): MaterialSubmissionHook {
  return {
    async prepare(event) {
      const activePrepare = preparing.get(event.requestId)
      if (activePrepare) {
        if (activePrepare.sessionId === event.sessionId) {
          await activePrepare.promise
          return
        }
        throw new Error('这次提交的材料范围正在另一个会话准备，已停止发送。')
      }
      const existing = reservations.get(event.requestId)
      if (existing) {
        if (existing.sessionId === event.sessionId) return
        throw new Error('这次提交的材料范围已绑定到另一个会话，已停止发送。')
      }

      const root = cwdOf(event.sessionId)
      const caseRef = root ? app.get().cases.find((candidate) => samePath(candidate.root, root)) : undefined
      const block = caseRef ? materialBatchBlockFor(event.sessionId, caseRef.case_id) : undefined
      if (block) throw new Error(materialBatchBlockText(block.reason))
      const batch = caseRef ? app.get().materialBatches[event.sessionId] : undefined
      if (!batch) return
      if (!caseRef || batch.case_id !== caseRef.case_id) {
        clearMaterialBatch(event.sessionId, batch.batch_id)
        throw new Error('案件已经切换，原批次已经取消。请重新导入材料后再发送。')
      }
      if (event.mode === 'steer') {
        throw new Error('本批材料只能随新的一问发送。本条是插话，本轮任务范围开始后不能再扩大；请等本轮结束后发送，或先取消本批。')
      }
      if (batch.selected_ids.length === 0) {
        clearMaterialBatch(event.sessionId, batch.batch_id)
        return
      }

      const occupiedKey = sessionKey(caseRef.case_id, event.sessionId)
      if (occupiedSessions.has(occupiedKey)) {
        throw new Error('上一批材料正在随另一条消息准备或发送，请稍后再发。')
      }
      occupiedSessions.set(occupiedKey, event.requestId)

      const operation = (async (): Promise<void> => {
        let releaseNeeded = true
        try {
          const r = await call<{
            action: 'prepare'
            request_id: string
            case_id: string
            session_id: string
            material_ids: string[]
          }>('requestScope', {
            action: 'prepare',
            case_id: caseRef.case_id,
            session_id: event.sessionId,
            request_id: event.requestId,
            material_ids: batch.selected_ids,
          })
          if (!r.ok) throw new Error(errorText(r.error))
          if (r.value.action !== 'prepare' || r.value.request_id !== event.requestId ||
            r.value.case_id !== caseRef.case_id || r.value.session_id !== event.sessionId ||
            !sameIds(r.value.material_ids, batch.selected_ids)) {
            throw new Error('材料范围没有按本次提交正确绑定，已停止发送。')
          }
          const currentRoot = cwdOf(event.sessionId)
          const currentCase = currentRoot
            ? app.get().cases.find((candidate) => samePath(candidate.root, currentRoot))
            : undefined
          const currentBatch = currentCase
            ? app.get().materialBatches[event.sessionId]
            : undefined
          const currentBlock = currentCase
            ? materialBatchBlockFor(event.sessionId, currentCase.case_id)
            : undefined
          if (!currentCase ||
            currentCase.case_id !== caseRef.case_id ||
            !samePath(currentCase.root, caseRef.root) ||
            currentBlock !== undefined ||
            !currentBatch ||
            currentBatch.case_id !== currentCase.case_id ||
            currentBatch.batch_id !== batch.batch_id ||
            !sameIds(currentBatch.selected_ids, batch.selected_ids)) {
            throw new Error('本次材料批次、选择或案件已经改变，原提交已停止，请重新发送当前批次。')
          }
          reservations.set(event.requestId, { ...event, batch })
          releaseNeeded = false
        } finally {
          // 服务可能已经准备但响应在网络里丢失；任何未确认成功的准备都尝试释放同一个 R。
          if (releaseNeeded) {
            await call('requestScope', {
              action: 'release',
              case_id: caseRef.case_id,
              session_id: event.sessionId,
              request_id: event.requestId,
            })
          }
        }
      })()
      preparing.set(event.requestId, { sessionId: event.sessionId, promise: operation })
      try {
        await operation
      } catch (error) {

        throw error
      } finally {

      }
    },
