0664 function isMainModule() {
0665   if (!process.argv[1]) return false;
0666   const moduleUrl = new URL(import.meta.url);
0667   if (moduleUrl.protocol !== "file:") return false;
0668   let modulePath = decodeURIComponent(moduleUrl.pathname);
0669   if (process.platform === "win32" && /^\/[A-Za-z]:\//.test(modulePath)) modulePath = modulePath.slice(1);
0670   const invoked = path.resolve(process.argv[1]);
0671   const current = path.normalize(modulePath);
0672   return process.platform === "win32"
0673     ? invoked.toLowerCase() === current.toLowerCase()
0674     : invoked === current;
0675 }
