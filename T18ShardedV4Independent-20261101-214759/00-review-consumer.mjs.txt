0911 async function readCredentialFromStdin() {
0912   if (process.stdin.isTTY) fail("credential", "TTY_NOT_ALLOWED");
0913   return new Promise((resolve, reject) => {
0914     let captured = Buffer.alloc(0);
0915     let settled = false;
0916     const cleanup = () => {
0917       process.stdin.off("data", onData);
0918       process.stdin.off("end", onEnd);
0919       process.stdin.off("error", onError);
0920       if (typeof process.stdin.pause === "function") process.stdin.pause();
0921     };
0922     const finish = () => {
0923       if (settled) return;
0924       settled = true;
0925       cleanup();
0926       let line = captured;
0927       if (line.length > 0 && line[line.length - 1] === 13) line = line.subarray(0, -1);
0928       if (line.length === 0 || line.length > 4096 ||
0929           [...line].some((byte) => byte < 33 || byte > 126)) {
0930         reject(new PilotProtocolError("credential", "CREDENTIAL_INVALID"));
0931         return;
0932       }
0933       resolve(line.toString("ascii"));
0934     };
0935     const onData = (chunk) => {
0936       if (settled) return;
0937       const bytes = Buffer.from(chunk);
0938       captured = Buffer.concat([captured, bytes]);
0939       const newline = captured.indexOf(10);
0940       if (newline >= 0) {
0941         captured = captured.subarray(0, newline);
0942         finish();
0943       } else if (captured.length > 4096) {
0944         settled = true;
0945         cleanup();
0946         reject(new PilotProtocolError("credential", "CREDENTIAL_INVALID"));
0947       }
0948     };
0949     const onEnd = () => finish();
0950     const onError = () => {
0951       if (settled) return;
0952       settled = true;
0953       cleanup();
0954       reject(new PilotProtocolError("credential", "STDIN_FAILED"));
0955     };
0956     process.stdin.on("data", onData);
0957     process.stdin.on("end", onEnd);
0958     process.stdin.on("error", onError);
0959     process.stdin.resume();
0960   });

1008 export async function performRequest(stage, requestBody, credential, deadline) {
1009   const startedAt = Date.now();
1010   const timeoutMs = requestTimeoutForDeadline(startedAt, deadline);
1011   if (timeoutMs <= 0) fail(stage, "ENTIRE_CALL_CAP_EXCEEDED");
1012   const signal = AbortSignal.timeout(timeoutMs);
1013   let responseStatus = null;
1014   const chunks = [];
1015   let capturedBytes = 0;
1016   let overflow = false;
1017   try {
1018     const response = await fetch(ROUTE, {
1019       method: "POST",
1020       redirect: "error",
1021       headers: { Authorization: buildAuthorizationHeader(credential), "Content-Type": "application/json" },
1022       body: requestBody,
1023       signal,
1024     });
1025     responseStatus = response.status;
1026     if (response.body) {
1027       const reader = response.body.getReader();
1028       let streamFailed = false;
1029       try {
1030         while (true) {
1031           const { done, value } = await reader.read();
1032           if (done) break;
1033           const chunk = Buffer.from(value);
1034           if (capturedBytes + chunk.byteLength > RESPONSE_CAP_BYTES) {
1035             overflow = true;
1036             await reader.cancel();
1037             break;
1038           }
1039           chunks.push(chunk);
1040           capturedBytes += chunk.byteLength;
1041         }
1042       } catch (error) {
1043         streamFailed = true;
1044         throw error;
1045       } finally {
1046         if (streamFailed) {
1047           try {
1048             await reader.cancel();
1049           } catch {
1050             // Cancel errors are swallowed only for a stream that already failed.
1051           }
1052         }
1053         reader.releaseLock();
1054       }
1055     }
1056     return {
1057       status: response.status,
1058       body: Buffer.concat(chunks).toString("utf8"),
1059       capturedBytes,
1060       responseOverflow: overflow,
1061       durationMs: Date.now() - startedAt,
1062       transportError: null,
1063       transportDiagnostic: null,
1064     };
1065   } catch (error) {
1066     const signalTimedOut = signal.aborted === true &&
1067       signal.reason && signal.reason.name === "TimeoutError";
1068     const timedOut = (error && error.name === "TimeoutError") || signalTimedOut;
1069     const wholeDeadlineExpired = Date.now() >= deadline ||
1070       (timedOut && timeoutMs < PER_CALL_TIMEOUT_MS);
1071     return {
1072       status: responseStatus,
1073       body: Buffer.concat(chunks).toString("utf8"),
1074       capturedBytes,
1075       responseOverflow: overflow,
1076       durationMs: Date.now() - startedAt,
1077       transportError: wholeDeadlineExpired
1078         ? "ENTIRE_CALL_CAP_EXCEEDED"
1079         : timedOut ? "TIMEOUT" : "TRANSPORT_FAILURE",
1080       transportDiagnostic: transportDiagnosticForError(error),
1081     };
1082   }
1083 }

1085 async function writeFailure(artifactsDir, credential, error, runStartedAt, timings) {
1086   const safeCode = error instanceof PilotProtocolError && /^[A-Z0-9_]+$/.test(error.code)
1087     ? error.code
1088     : "INTERNAL_FAILURE";
1089   const stage = error instanceof PilotProtocolError && /^[a-z0-9_]+$/.test(error.stage)
1090     ? error.stage
1091     : "runtime";
1092   const status = `FAIL_${safeCode}`;
1093   await writeArtifact(artifactsDir, "error.json", { stage, code: safeCode }, credential);
1094   await writeArtifact(artifactsDir, "timings.json", {
1095     ...timings,
1096     totalMs: Date.now() - runStartedAt,
1097   }, credential);
1098   await writeArtifact(artifactsDir, "final-report.json", {
1099     status,
1100     protocol_capability_only: true,
1101     legal_or_business_pass: false,
1102     stage,
1103     code: safeCode,
1104   }, credential);
1105   return { status, stage, code: safeCode };
1106 }

1108 async function runMain() {

1110   const deadline = runStartedAt + ENTIRE_CALL_CAP_MS;
1111   let artifactsDir = null;
1112   let credential = "";
1113   let timings = {};

1124     const units = buildVisibleUnits(input.receipts);
1125     const stage1Parsed = parsePinnedStage1Assessments(pinned.stage1CacheBytes, units);

1141       entireCallCapMs: ENTIRE_CALL_CAP_MS,

1163       reusedStage1: true,
1164       sourceCacheIdentity: pinned.sourceCacheIdentity,
1165       assessments: stage1Parsed.assessments,
1166     }, credential);
1167     await writeArtifact(artifactsDir, "shard-plan.json", {
1168       targetCount,
1169       shardCount: shardPlan.length,
1170       shards: shardPlan,
1171     }, credential);
1172 
1173     const shardReviewSets = [];
1174     for (const shard of shardPlan) {
1175       const stage = shard.shard_id;
1176       if (Date.now() >= deadline) fail(stage, "ENTIRE_CALL_CAP_EXCEEDED");
1177       const stage2Payload = buildRequestPayload(
1178         buildStage2Messages(
1179           units,
1180           stage1Parsed.assessments,
1181           inventory,
1182           input.actualCoverage,
1183           shard.review_target_ids,
1184         ),
1185       );
1186       const stage2Body = JSON.stringify(stage2Payload);
1187       await writeArtifact(artifactsDir, `${stage}-request.json`, { body: stage2Body }, credential);
1188       await writeArtifact(artifactsDir, `${stage}-digests.json`, {
1189         messages: digest(JSON.stringify(stage2Payload.messages)),
1190         requestBody: digest(stage2Body),
1191       }, credential);
1192       const stage2Response = await performRequest(stage, stage2Body, credential, deadline);
1193       timings = { ...timings, [`${stage}Ms`]: stage2Response.durationMs };
1194       await writeArtifact(artifactsDir, `${stage}-response.json`, {
1195         status: stage2Response.status,
1196         body: stage2Response.body,
1197         bodyOverflow: stage2Response.responseOverflow,
1198         capturedBytes: stage2Response.capturedBytes,
1199         transportError: stage2Response.transportError,
1200         transportDiagnostic: stage2Response.transportDiagnostic,
1201       }, credential);
1202       if (Date.now() >= deadline) fail(stage, "ENTIRE_CALL_CAP_EXCEEDED");
1203       if (stage2Response.transportError === "ENTIRE_CALL_CAP_EXCEEDED") {
1204         fail(stage, "ENTIRE_CALL_CAP_EXCEEDED");
1205       }
1206       if (stage2Response.transportError === "TIMEOUT") fail(stage, "TIMEOUT");
1207       if (stage2Response.transportError) fail(stage, "TRANSPORT_FAILURE");
1208       if (stage2Response.status !== 200) fail(stage, "HTTP_NON_200");
1209       if (stage2Response.responseOverflow) fail(stage, "RESPONSE_TOO_LARGE");
1210       const stage2Parsed = validateStage2Result(
1211         parseModelJson(unwrapChatCompletion(stage2Response.body)),
1212         inventory,
1213         shard.review_target_ids,
1214       );
1215       await writeArtifact(artifactsDir, `${stage}-parsed.json`, stage2Parsed, credential);
1216       shardReviewSets.push(stage2Parsed.reviews);
1217     }
1218     const stage2Parsed = mergeShardReviews(inventory, shardPlan, shardReviewSets);
1219     await writeArtifact(artifactsDir, "stage2-parsed.json", stage2Parsed, credential);
1220     const status = outputStatusForReviews(stage2Parsed.reviews);
1221     timings = { ...timings, totalMs: Date.now() - runStartedAt };
1222     await writeArtifact(artifactsDir, "timings.json", timings, credential);
1223     await writeArtifact(artifactsDir, "final-report.json", {
1224       status,
1225       protocol_capability_only: true,
1226       legal_or_business_pass: false,
1227       reusedStage1: true,
1228       sourceCacheIdentity: pinned.sourceCacheIdentity,
1229       sourceUnitCount: units.length,
1230       sourceTargetCount: inventory.sourceTargets.length,
1231       draftTargetCount: inventory.draftTargets.length,
1232       selfcheckTargetCount: inventory.selfcheckTargets.length,
1233       stage1AssessmentReuseCount: stage1Parsed.assessments.length,
1234       stage2ReviewCount: stage2Parsed.reviews.length,
1235       targetCount,
1236       shardCount: shardPlan.length,
1237       callsMade: 2,
1238     }, credential);
1239     process.stdout.write(`status=${status}\nreport=${redactCredential(path.join(artifactsDir, "final-report.json"), credential)}\n`);
1240     return 0;
1241   } catch (error) {
1242     if (artifactsDir) {
1243       try {
1244         const failure = await writeFailure(artifactsDir, credential, error, runStartedAt, timings);
1245         process.stderr.write(`status=${failure.status}\nstage=${failure.stage}\n`);
1246       } catch {
1247         process.stderr.write("status=FAIL_ARTIFACT_WRITE\nstage=output\n");
1248       }
1249     } else {
1250       const code = error instanceof PilotProtocolError && /^[A-Z0-9_]+$/.test(error.code)
1251         ? error.code
1252         : "INTERNAL_FAILURE";
1253       const stage = error instanceof PilotProtocolError && /^[a-z0-9_]+$/.test(error.stage)
1254         ? error.stage
1255         : "runtime";
1256       process.stderr.write(`status=FAIL_${code}\nstage=${stage}\n`);
1257     }
1258     return 1;
1259   }

1262 function isMainModule() {
1263   if (!process.argv[1]) return false;
1264   const moduleUrl = new URL(import.meta.url);
1265   if (moduleUrl.protocol !== "file:") return false;
1266   let modulePath = decodeURIComponent(moduleUrl.pathname);
1267   if (process.platform === "win32" && /^\/[A-Za-z]:\//.test(modulePath)) modulePath = modulePath.slice(1);
1268   const invoked = path.resolve(process.argv[1]);
1269   const current = path.normalize(modulePath);
1270   return process.platform === "win32"
1271     ? invoked.toLowerCase() === current.toLowerCase()
1272     : invoked === current;
1273 }
