0032 function isLoopbackHost(host) {
0033   const value = normalizedHost(host)

0038 function parseDestination(value) {
0039   if (value instanceof URL) return { protocol: value.protocol, host: value.hostname, origin: value.origin, url: value }
0040   if (typeof value === 'string') {
0041     try {
0042       const parsed = new URL(value)
0043       return { protocol: parsed.protocol, host: parsed.hostname, origin: parsed.origin, url: parsed }
0044     } catch {}
0045   }
0046   if (value && typeof value === 'object') {
0047     const protocol = String(value.protocol || 'http:').replace(/:$/, '') + ':'
0048     const host = value.hostname || value.host || 'localhost'
0049     return { protocol, host, origin: `${protocol}//${host}`, url: undefined }
0050   }
0051   return { protocol: 'unknown:', host: '', origin: 'unknown', url: undefined }
0052 }

0108 function socketDestinations(args) {
0109   if (args.length === 0) return
0110   const first = args[0]
0111   if (typeof first === 'string') return undefined
0112   if (first instanceof URL) return [parseDestination(first)]
0113   if (first && typeof first === 'object') {
0114     if (typeof first.path === 'string' && first.path.length > 0) {
0115       return undefined
0116     }
0117     const hosts = [first.host, first.hostname]
0118       .filter((host) => host !== undefined && host !== null && String(host) !== '')
0119       .map(String)
0120     if (hosts.length === 0) hosts.push('localhost')
0121     return hosts.map((host) => ({ protocol: 'unknown:', host, origin: `tcp://${host}`, url: undefined }))
0122   }
0123   if (typeof first === 'number') {
0124     const second = args[1]
0125     let host = 'localhost'
0126     if (typeof second === 'string' && second !== '') {
0127       host = second
0128     } else if (second !== undefined && typeof second !== 'function') {
0129       return [{ protocol: 'unknown:', host: '[invalid numeric host]', origin: 'tcp://invalid', url: undefined }]
0130     }
0131     return [{ protocol: 'unknown:', host, origin: `tcp://${host}`, url: undefined }]
0132   }
0133   return [{ protocol: 'unknown:', host: '', origin: 'tcp://invalid', url: undefined }]
