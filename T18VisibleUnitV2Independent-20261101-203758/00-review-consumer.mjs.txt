0504 async function verifyPinnedSources() {
0505   let manifest;
0506   try {
0507     manifest = JSON.parse(await fs.readFile(new URL(MANIFEST_NAME, import.meta.url), "utf8"));
0508   } catch {
0509     fail("source", "SOURCE_MANIFEST_INVALID");
0510   }
0511   if (!manifest || manifest.algorithm !== "sha256") fail("source", "SOURCE_MANIFEST_INVALID");
0512   const pins = [
0513     ["consumer", CONSUMER_NAME, manifest.consumer],
0514     ["fixed_data", FIXED_DATA_NAME, manifest.fixed_data],
0515     ["offline_controls", CONTROLS_NAME, manifest.offline_controls],
0516   ];
0517   let fixedDataBytes;
0518   for (const [label, fileName, pin] of pins) {
0519     if (!pin || pin.path !== fileName || !Number.isSafeInteger(pin.bytes) ||
0520         pin.bytes < 0 || !/^[0-9a-f]{64}$/.test(pin.sha256)) {
0521       fail("source", "SOURCE_MANIFEST_INVALID");
0522     }
0523     const bytes = await fs.readFile(new URL(fileName, import.meta.url));
0524     const actual = { bytes: bytes.byteLength, sha256: crypto.createHash("sha256").update(bytes).digest("hex") };
0525     if (actual.bytes !== pin.bytes || actual.sha256 !== pin.sha256) fail("source", `${label.toUpperCase()}_PIN_MISMATCH`);
0526     if (label === "fixed_data") fixedDataBytes = bytes;
0527   }
0528   return fixedDataBytes.toString("utf8");
0529 }

0569 async function readCredentialFromStdin() {
0570   if (process.stdin.isTTY) fail("credential", "TTY_NOT_ALLOWED");
0571   return new Promise((resolve, reject) => {
0572     let captured = Buffer.alloc(0);
0573     let settled = false;
0574     const cleanup = () => {
0575       process.stdin.off("data", onData);
0576       process.stdin.off("end", onEnd);
0577       process.stdin.off("error", onError);
0578       if (typeof process.stdin.pause === "function") process.stdin.pause();
0579     };
0580     const finish = () => {
0581       if (settled) return;
0582       settled = true;
0583       cleanup();
0584       let line = captured;
0585       if (line.length > 0 && line[line.length - 1] === 13) line = line.subarray(0, -1);
0586       if (line.length === 0 || line.length > 4096 ||
0587           [...line].some((byte) => byte < 33 || byte > 126)) {
0588         reject(new PilotProtocolError("credential", "CREDENTIAL_INVALID"));
0589         return;
0590       }
0591       resolve(line.toString("ascii"));
0592     };
0593     const onData = (chunk) => {
0594       if (settled) return;
0595       const bytes = Buffer.from(chunk);
0596       captured = Buffer.concat([captured, bytes]);
0597       const newline = captured.indexOf(10);
0598       if (newline >= 0) {
0599         captured = captured.subarray(0, newline);
0600         finish();
0601       } else if (captured.length > 4096) {
0602         settled = true;
0603         cleanup();
0604         reject(new PilotProtocolError("credential", "CREDENTIAL_INVALID"));
0605       }
0606     };
0607     const onEnd = () => finish();
0608     const onError = () => {
0609       if (settled) return;
0610       settled = true;
0611       cleanup();
0612       reject(new PilotProtocolError("credential", "STDIN_FAILED"));
0613     };
0614     process.stdin.on("data", onData);
0615     process.stdin.on("end", onEnd);
0616     process.stdin.on("error", onError);
0617     process.stdin.resume();
0618   });
0619 }

0621 export async function performRequest(stage, requestBody, credential, deadline) {
0622   const startedAt = Date.now();
0623   const timeoutMs = requestTimeoutForDeadline(startedAt, deadline);
0624   if (timeoutMs <= 0) fail(stage, "ENTIRE_CALL_CAP_EXCEEDED");
0625   const signal = AbortSignal.timeout(timeoutMs);
0626   let responseStatus = null;
0627   const chunks = [];
0628   let capturedBytes = 0;
0629   let overflow = false;
0630   try {
0631     const response = await fetch(ROUTE, {
0632       method: "POST",
0633       redirect: "error",
0634       headers: { Authorization: buildAuthorizationHeader(credential), "Content-Type": "application/json" },
0635       body: requestBody,
0636       signal,
0637     });
0638     responseStatus = response.status;
0639     if (response.body) {
0640       const reader = response.body.getReader();
0641       let streamFailed = false;
0642       try {
0643         while (true) {
0644           const { done, value } = await reader.read();
0645           if (done) break;
0646           const chunk = Buffer.from(value);
0647           if (capturedBytes + chunk.byteLength > RESPONSE_CAP_BYTES) {
0648             overflow = true;
0649             await reader.cancel();
0650             break;
0651           }
0652           chunks.push(chunk);
0653           capturedBytes += chunk.byteLength;
0654         }
0655       } catch (error) {
0656         streamFailed = true;
0657         throw error;
0658       } finally {
0659         if (streamFailed) {
0660           try {
0661             await reader.cancel();
0662           } catch {
0663             // Cancel errors are swallowed only for a stream that already failed.
0664           }
0665         }
0666         reader.releaseLock();
0667       }
0668     }
0669     return {
0670       status: response.status,
0671       body: Buffer.concat(chunks).toString("utf8"),
0672       capturedBytes,
0673       responseOverflow: overflow,
0674       durationMs: Date.now() - startedAt,
0675       transportError: null,
0676     };
0677   } catch (error) {
0678     const signalTimedOut = signal.aborted === true &&
0679       signal.reason && signal.reason.name === "TimeoutError";
0680     const timedOut = (error && error.name === "TimeoutError") || signalTimedOut;
0681     const wholeDeadlineExpired = Date.now() >= deadline ||
0682       (timedOut && timeoutMs < PER_CALL_TIMEOUT_MS);
0683     return {
0684       status: responseStatus,
0685       body: Buffer.concat(chunks).toString("utf8"),
0686       capturedBytes,
0687       responseOverflow: overflow,
0688       durationMs: Date.now() - startedAt,
0689       transportError: wholeDeadlineExpired
0690         ? "ENTIRE_CALL_CAP_EXCEEDED"
0691         : timedOut ? "TIMEOUT" : "TRANSPORT_FAILURE",
0692     };
0693   }
0694 }

0857 function isMainModule() {
0858   if (!process.argv[1]) return false;
0859   const moduleUrl = new URL(import.meta.url);
0860   if (moduleUrl.protocol !== "file:") return false;
0861   let modulePath = decodeURIComponent(moduleUrl.pathname);
0862   if (process.platform === "win32" && /^\/[A-Za-z]:\//.test(modulePath)) modulePath = modulePath.slice(1);
0863   const invoked = path.resolve(process.argv[1]);
0864   const current = path.normalize(modulePath);
0865   return process.platform === "win32"
0866     ? invoked.toLowerCase() === current.toLowerCase()
0867     : invoked === current;
0868 }
