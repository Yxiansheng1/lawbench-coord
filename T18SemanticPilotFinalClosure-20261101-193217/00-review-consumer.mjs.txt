0537       responseOverflow: overflow,
0538       durationMs: Date.now() - startedAt,
0539       transportError: null,
0540     };
0541   } catch (error) {
0542     const signalTimedOut = signal.aborted === true &&
0543       signal.reason && signal.reason.name === "TimeoutError";
0544     const timedOut = (error && error.name === "TimeoutError") || signalTimedOut;
0545     const wholeDeadlineExpired = Date.now() >= deadline ||
0546       (timedOut && timeoutMs < PER_CALL_TIMEOUT_MS);
0547     return {
0548       status: responseStatus,
0549       body: Buffer.concat(chunks).toString("utf8"),
0550       capturedBytes,
0551       responseOverflow: overflow,
0552       durationMs: Date.now() - startedAt,
0553       transportError: wholeDeadlineExpired
0554         ? "ENTIRE_CALL_CAP_EXCEEDED"
0555         : timedOut ? "TIMEOUT" : "TRANSPORT_FAILURE",
0556     };
0557   }
0558 }

0560 async function writeFailure(artifactsDir, credential, error, runStartedAt, timings) {
0561   const safeCode = error instanceof PilotProtocolError && /^[A-Z0-9_]+$/.test(error.code)
0562     ? error.code
0563     : "INTERNAL_FAILURE";
0564   const stage = error instanceof PilotProtocolError && /^[a-z0-9_]+$/.test(error.stage)
0565     ? error.stage
0566     : "runtime";
0567   const status = `FAIL_${safeCode}`;
0568   await writeArtifact(artifactsDir, "error.json", { stage, code: safeCode }, credential);
0569   await writeArtifact(artifactsDir, "timings.json", {
0570     ...timings,
0571     totalMs: Date.now() - runStartedAt,
0572   }, credential);
0573   await writeArtifact(artifactsDir, "final-report.json", {
0574     status,
0575     protocol_capability_only: true,
0576     legal_or_business_pass: false,
0577     stage,
0578     code: safeCode,
0579   }, credential);
0580   return { status, stage, code: safeCode };
0581 }

0583 async function runMain() {
0584   const runStartedAt = Date.now();
0585   const deadline = runStartedAt + ENTIRE_CALL_CAP_MS;
0586   let artifactsDir = null;
0587   let credential = "";
0588   let timings = {};
0589   try {
0590     const args = parseArguments(process.argv.slice(2));
0591     const fixedData = await verifyPinnedSources();
0592     let data;
0593     try {
0594       data = JSON.parse(fixedData);
0595     } catch {
0596       fail("input", "FIXED_DATA_INVALID");
0597     }
0598     const input = selectPilotInput(data);
0599     artifactsDir = await prepareOutputRoot(args.root);
0600     credential = await readCredentialFromStdin();
0601     await writeArtifact(artifactsDir, "route.json", {
0602       method: "POST",
0603       url: ROUTE,
0604       redirect: "error",
0605       model: MODEL,
0606       parameters: PARAMETERS,
0607       maxSequentialCalls: MAX_SEQUENTIAL_CALLS,
0608       perCallTimeoutMs: PER_CALL_TIMEOUT_MS,
0609       entireCallCapMs: ENTIRE_CALL_CAP_MS,
0610       responseCapBytes: RESPONSE_CAP_BYTES,
0611       requestHeadersPersisted: false,
0612     }, credential);
0613 
0614     const stage1Payload = buildRequestPayload(buildStage1Messages(input));
0615     const stage1Body = JSON.stringify(stage1Payload);
0616     await writeArtifact(artifactsDir, "stage1-request.json", { body: stage1Body }, credential);
0617     await writeArtifact(artifactsDir, "stage1-digests.json", {
0618       messages: digest(JSON.stringify(stage1Payload.messages)),
0619       requestBody: digest(stage1Body),
0620     }, credential);
0621     const stage1Response = await performRequest("stage1", stage1Body, credential, deadline);
0622     await writeArtifact(artifactsDir, "stage1-response.json", {
0623       status: stage1Response.status,
0624       body: stage1Response.body,
0625       bodyOverflow: stage1Response.responseOverflow,
0626       capturedBytes: stage1Response.capturedBytes,
0627       transportError: stage1Response.transportError,
0628     }, credential);
0629     if (Date.now() >= deadline) fail("stage1", "ENTIRE_CALL_CAP_EXCEEDED");
0630     timings = { stage1Ms: stage1Response.durationMs };
0631     if (stage1Response.transportError === "ENTIRE_CALL_CAP_EXCEEDED") {
0632       fail("stage1", "ENTIRE_CALL_CAP_EXCEEDED");
0633     }
0634     if (stage1Response.transportError === "TIMEOUT") fail("stage1", "TIMEOUT");
0635     if (stage1Response.transportError) fail("stage1", "TRANSPORT_FAILURE");
0636     if (stage1Response.status !== 200) fail("stage1", "HTTP_NON_200");
0637     if (stage1Response.responseOverflow) fail("stage1", "RESPONSE_TOO_LARGE");
0638     const stage1Parsed = validateStage1Result(
0639       parseModelJson(unwrapChatCompletion(stage1Response.body)),
0640       input.receipts,
0641     );
0642     await writeArtifact(artifactsDir, "stage1-parsed.json", stage1Parsed, credential);
0643 
0644     const stage2Payload = buildRequestPayload(buildStage2Messages(input, stage1Parsed.facts));
0645     const stage2Body = JSON.stringify(stage2Payload);
0646     await writeArtifact(artifactsDir, "stage2-request.json", { body: stage2Body }, credential);
0647     await writeArtifact(artifactsDir, "stage2-digests.json", {
0648       messages: digest(JSON.stringify(stage2Payload.messages)),
0649       requestBody: digest(stage2Body),
0650     }, credential);
0651     const stage2Response = await performRequest("stage2", stage2Body, credential, deadline);
0652     await writeArtifact(artifactsDir, "stage2-response.json", {
0653       status: stage2Response.status,
0654       body: stage2Response.body,
0655       bodyOverflow: stage2Response.responseOverflow,
0656       capturedBytes: stage2Response.capturedBytes,
0657       transportError: stage2Response.transportError,
0658     }, credential);
0659     if (Date.now() >= deadline) fail("stage2", "ENTIRE_CALL_CAP_EXCEEDED");
0660     timings = { ...timings, stage2Ms: stage2Response.durationMs };
0661     if (stage2Response.transportError === "ENTIRE_CALL_CAP_EXCEEDED") {
0662       fail("stage2", "ENTIRE_CALL_CAP_EXCEEDED");
0663     }
0664     if (stage2Response.transportError === "TIMEOUT") fail("stage2", "TIMEOUT");
0665     if (stage2Response.transportError) fail("stage2", "TRANSPORT_FAILURE");
0666     if (stage2Response.status !== 200) fail("stage2", "HTTP_NON_200");
0667     if (stage2Response.responseOverflow) fail("stage2", "RESPONSE_TOO_LARGE");
0668     const stage2Parsed = validateStage2Result(parseModelJson(unwrapChatCompletion(stage2Response.body)), {
0669       receipts: input.receipts,
0670       draft: input.draft,
0671       visibleSelfcheck: input.visibleSelfcheck,
0672       stage1Candidates: stage1Parsed.facts,
0673     });
0674     await writeArtifact(artifactsDir, "stage2-parsed.json", stage2Parsed, credential);
0675     const status = outputStatusForFindings(stage2Parsed.findings);
0676     timings = { ...timings, totalMs: Date.now() - runStartedAt };
0677     await writeArtifact(artifactsDir, "timings.json", timings, credential);
0678     await writeArtifact(artifactsDir, "final-report.json", {
0679       status,
0680       protocol_capability_only: true,
0681       legal_or_business_pass: false,
0682       stage1CandidateCount: stage1Parsed.facts.length,
0683       stage2FindingCount: stage2Parsed.findings.length,
0684       callsMade: 2,
0685     }, credential);
0686     process.stdout.write(`status=${status}\nreport=${redactCredential(path.join(artifactsDir, "final-report.json"), credential)}\n`);
0687     return 0;
0688   } catch (error) {
0689     if (artifactsDir) {
0690       try {
0691         const failure = await writeFailure(artifactsDir, credential, error, runStartedAt, timings);
0692         process.stderr.write(`status=${failure.status}\nstage=${failure.stage}\n`);
0693       } catch {
0694         process.stderr.write("status=FAIL_ARTIFACT_WRITE\nstage=output\n");
0695       }
0696     } else {
0697       const code = error instanceof PilotProtocolError && /^[A-Z0-9_]+$/.test(error.code)
0698         ? error.code
0699         : "INTERNAL_FAILURE";
0700       const stage = error instanceof PilotProtocolError && /^[a-z0-9_]+$/.test(error.stage)
0701         ? error.stage
0702         : "runtime";
0703       process.stderr.write(`status=FAIL_${code}\nstage=${stage}\n`);
0704     }
0705     return 1;
0706   }
0707 }

0709 function isMainModule() {
0710   if (!process.argv[1]) return false;
0711   const moduleUrl = new URL(import.meta.url);
0712   if (moduleUrl.protocol !== "file:") return false;
0713   let modulePath = decodeURIComponent(moduleUrl.pathname);
0714   if (process.platform === "win32" && /^\/[A-Za-z]:\//.test(modulePath)) modulePath = modulePath.slice(1);
0715   const invoked = path.resolve(process.argv[1]);
0716   const current = path.normalize(modulePath);
0717   return process.platform === "win32"
0718     ? invoked.toLowerCase() === current.toLowerCase()
0719     : invoked === current;
0720 }
0721 
0722 if (isMainModule()) {
0723   runMain().then((code) => {
0724     process.exitCode = code;
0725   });
