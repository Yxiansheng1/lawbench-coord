/** 一段正常的长回答：阅卷要点，各条内容不同，表格行的格式一样。 */
function longAnswer(): string {
  const rows = Array.from({ length: 60 }, (_, i) => `| ${i + 1} | 2025年${(i % 12) + 1}月${(i % 28) + 1}日 | 张某甲以第${i + 1}种名义收取被害人款项人民币${(i + 1) * 1371}元 | 〔起诉意见书 第${(i % 5) + 1}页〕 |`)

}

/** 按流式输出的样子切成小段（每段 1–7 个字）。 */
function* deltas(text: string): Generator<string> {
  let i = 0, k = 0
  while (i < text.length) { const n = 1 + (k++ * 5) % 7; yield text.slice(i, i + n); i += n }
}

describe('① 思考 / 回复原样重复（RepetitionGuard）', () => {
  it(`同一段（≥${SEGMENT} 字）第 ${REPEAT_LIMIT} 次出现才判重复；两次不判`, () => {
    expect([SEGMENT, REPEAT_LIMIT]).toEqual([80, 3])
    expect(LOOPED.replace(/\s/g, '').length).toBeGreaterThan(SEGMENT)
    const g = new RepetitionGuard()
    let at = -1, n = 0
    for (const d of deltas(LOOPED + LOOPED + LOOPED)) { n += d.length; if (g.push('reasoning', d)) { at = n; break } }
    // 第三遍开头 80 字处判出来，不等整段写完
    // 位置按去空白后的字数算
    const L = LOOPED.replace(/\s/g, '').length
    const seen = (LOOPED + LOOPED + LOOPED).slice(0, at).replace(/\s/g, '').length
    expect(seen).toBeGreaterThanOrEqual(L * 2 + SEGMENT)
    expect(seen).toBeLessThan(L * 2 + SEGMENT + 7)
    const two = new RepetitionGuard()
    for (const d of deltas(LOOPED + LOOPED)) expect(two.push('reasoning', d)).toBe(false)
  })

  it('去空白比较：换行、空格不同的重复照样判出来', () => {
    const g = new RepetitionGuard()
    const variants = [LOOPED, LOOPED.replace(/，/g, '， ').replace(/。/g, '。\n'), `\n\n${LOOPED.replace(/ /g, '')}`]
    expect(variants.map((v) => [...deltas(v)].some((d) => g.push('reasoning', d)))).toEqual([false, false, true])
  })

  it('正常长回答不误杀：表格行格式相同、内容不同；短句（不足 80 字）夹在不同内容之间重复多次也不算', () => {
    const g = new RepetitionGuard()
    for (const d of deltas(longAnswer())) expect(g.push('text', d)).toBe(false)
    const short = new RepetitionGuard()
    // 每条事实后面都跟一句同样的提醒（不足 80 字），夹在不同的内容之间
    for (let i = 0; i < 50; i++) expect(short.push('text', `第${i + 1}笔转账${(i + 3) * 997}元，时间2025年${(i % 12) + 1}月。请律师核对原件。出处：〔起诉意见书 第2页〕。`)).toBe(false)
    // 同一短句连着写成一长串：重复的单元不足 80 字、只有几种字，不算（复核 rv-A64 P2-1：与分隔线同一口径；见 repetition-samples.spec.ts）
    const run = new RepetitionGuard()
    expect(Array.from({ length: 30 }, () => run.push('text', '请律师核对原件。')).includes(true)).toBe(false)
  })

  it('思考和回复分开数：回答里复述一次思考里的话不算；判出后一直为真', () => {
    const g = new RepetitionGuard()
    expect(g.push('reasoning', LOOPED + LOOPED)).toBe(false)
    expect(g.push('text', LOOPED)).toBe(false)
    expect(g.push('reasoning', LOOPED)).toBe(true)
    expect(g.push('text', '好的')).toBe(true)
    expect(g.tripped).toBe(true)
  })

  it('同一个字连写不算（复核 rv-A64 P2-1：分隔线、"哈哈哈"一类不足 10 种字）', () => {
    const g = new RepetitionGuard()
    expect(g.push('reasoning', '哈'.repeat(SEGMENT * 3 - 1))).toBe(false)
    expect(g.push('reasoning', '哈'.repeat(SEGMENT * 10))).toBe(false)
  })
})

function fakeCore() {
  const calls: Array<{ cmd: string; body: Record<string, unknown> }> = []
  const saved = JSON.parse(readFileSync(join(__dirname, '..', '..', 'contracts', 'examples', 'tool_save_draft.result.json'), 'utf8')) as unknown
  const core = {
    call: async (cmd: string, body: Record<string, unknown>) => {
      calls.push({ cmd, body })
      if (cmd === 'task/begin') return { ok: true, value: { task_id: T, params: PARAMS, budget: { model_calls: 32, tool_calls: 48, minutes: 45 } } }
      if (cmd === 'context') return { ok: true, value: { l0: { text: '案件卡片' }, l1: { text: '任务输入', truncated: false, toc: [] } } }
      if (cmd === 'tool') return { ok: true, value: saved }
      return { ok: true, value: {} }
    },
  } as unknown as CoreClient
  return { core, calls }
}
const agentObj = (id: string) => ({ id, session: { header: { cwd: 'D:\\案件\\张某甲诈骗案' } } })
const enter = () => ({ kind: 'enter' as const, messages: [] })
/** Agent 循环发的请求：整个冻结、不带 purpose（见 isAgentLoopCall）。 */
const loop = (sessionId: string) => Object.freeze({ sessionId })
type Chunk = { type: string; index?: number; text?: string; reason?: unknown; blockType?: string }

function setup() {
  const logs: Array<[string, string, Record<string, unknown> | undefined]> = []
  const noted: Array<[string, string, string | undefined]> = []
  const f = fakeCore()
  const a = new LegalAgent(f.core, ((level: string, event: string, meta?: Record<string, unknown>) => { logs.push([level, event, meta]) }) as never,
    (id, code, taskId) => { noted.push([id, code, taskId]) })
  return { a, f, logs, noted }
}

/** 模拟 6000D 的流：先开一个思考块，按 deltas 吐思考；记下被拉了多少块、有没有被提前关掉（上游请求取消）。 */
function upstream(text: string, finishKind = 'stop') {
  const seen = { pulled: 0, closed: false }
  async function* gen(): AsyncGenerator<Chunk> {
    try {
      yield { type: 'block-start', index: 0, blockType: 'reasoning' }
      for (const d of deltas(text)) { seen.pulled++; yield { type: 'reasoning-delta', index: 0, text: d } }
      yield { type: 'block-end', index: 0 }
      yield { type: 'finish', reason: { kind: finishKind } }
    } finally { seen.closed = true }
  }
  return { gen: gen(), seen }
}

async function drain(it: AsyncIterable<unknown>): Promise<Chunk[]> {
  const out: Chunk[] = []
  for await (const c of it) out.push(c as Chunk)
  return out
}

describe('① 接在 DSH 的 llm/stream 上（Agent 插件 guardStream）', () => {
