+def _defense_unread_numeric_citation(tasks, root: str, task: dict,
+                                     cites: list[dict]) -> tuple[str, dict] | None:
+    """返回首个未被本任务当前版本完整读取的数值型出处。
+
+    只核对读取区间、材料单元结构和程序待识别占位，不检查材料中的事实内容。
+    """
+    tid = task["task_id"]
+    reads = tasks._read(root, tasks.rel(tid, "reads.json"), "files/reads.schema.json")["reads"]
+    by_material_unit: dict[tuple[str, str, str], list[tuple[int, int]]] = {}
+    for rec in reads:
+        key = (rec["material_id"], rec["material_version"], rec["unit"])
+        by_material_unit.setdefault(key, []).append((rec["from"], rec["to"]))
+    materials = None
+    pending_units: dict[tuple[str, str, str], set[int] | None] = {}
+    for cite in cites:
+        loc = cite["loc"]
+        unit = loc["unit"]
+        if unit not in ("page", "para", "line"):
+            continue
+        start = loc["from"]
+        end = loc.get("to", start)
+        if end < start:
+            return cite["material_id"], loc
+        next_unit = start
+        for read_start, read_end in sorted(
+                by_material_unit.get((cite["material_id"], cite["material_version"], unit), ())):
+            if read_end < next_unit:
+                continue
+            if read_start > next_unit:
+                break
+            next_unit = max(next_unit, read_end + 1)
+            if next_unit > end:
+                break
+        if next_unit <= end:
+            return cite["material_id"], loc
+        key = (cite["material_id"], cite["material_version"], unit)
+        if key not in pending_units:
+            if materials is None:
+                materials = tasks.index_for_task(task)["materials"]
+            material = next((m for m in materials
+                             if (m["material_id"], m["sha256"], m["unit"]) == key), None)
+            if material is None:
+                pending_units[key] = None
+            else:
+                try:
+                    text = texts.read_text(root, material)
+                except ApiError as exc:
+                    if exc.code != "MATERIAL_NOT_READY":
+                        raise
+                    pending_units[key] = None
+                else:
+                    pending_units[key] = {
+                        u.no for u in texts.split_units(text, unit)
+                        if u.text.strip() == texts.PENDING_OCR
+                    }
+        if pending_units[key] is None or any(
+                start <= no <= end for no in pending_units[key]):
+            return cite["material_id"], loc
+    return None
