0074: def _exact_number(text: str, *, allow_zero: bool) -> Fraction | None:
0075:     """只接受本检查声明支持的有限中文小整数或阿拉伯十进制数。"""
0076:     if text and all(char in _CN_DIGITS or char == "十" for char in text):
0077:         if text == "十":
0078:             value = Fraction(10)
0079:         elif len(text) == 1:
0080:             value = Fraction(_CN_DIGITS[text])
0081:         elif text.count("十") == 1:
0082:             tens, ones = text.split("十")
0083:             if len(tens) > 1 or len(ones) > 1:
0084:                 return None
0085:             value = Fraction((_CN_DIGITS[tens] if tens else 1) * 10
0086:                              + (_CN_DIGITS[ones] if ones else 0))
0087:         else:
0088:             return None
0089:     else:
0090:         try:
0091:             value = Fraction(Decimal(text))
0092:         except (InvalidOperation, ValueError):
0093:             return None
0094:     if value < 0 or (value == 0 and not allow_zero):
0095:         return None
0096:     return value

0099: def _rate_unit(segment: str, start: int, end: int) -> str | None:
0100:     """仅识别明确的日/月/年计率单位；缺失或含糊时不参与判断。"""
0101:     suffix = re.match(r"\s*(?:/|每)\s*(日|月|年)", segment[end:end + 6])
0102:     if suffix:
0103:         return suffix.group(1)
0104:     prefix = segment[max(0, start - 6):start]
0105:     prefix_unit = re.search(r"(日|月|年)(?:利率|费率)?(?:为|是)?$", prefix)
0106:     return prefix_unit.group(1) if prefix_unit else None

0109: def _segment_rates(segment: str) -> list[dict]:
0110:     rates = []
0111:     for match in _RATE_TOKEN_RE.finditer(segment):
0112:         denominator = {"百分之": 100, "千分之": 1000, "万分之": 10000}[match.group("denominator")]
0113:         numerator = _exact_number(match.group("numerator"), allow_zero=True)
0114:         if numerator is None:
0115:             continue
0116:         rates.append({
0117:             "value": numerator / denominator,
0118:             "unit": _rate_unit(segment, match.start(), match.end()),
0119:             "start": match.start(),
0120:             "end": match.end(),
0121:         })
0122:     for match in _PERCENT_TOKEN_RE.finditer(segment):
0123:         numerator = _exact_number(match.group("numerator"), allow_zero=True)
0124:         if numerator is None:
0125:             continue
0126:         rates.append({
0127:             "value": numerator / 100,
0128:             "unit": _rate_unit(segment, match.start(), match.end()),
0129:             "start": match.start(),
0130:             "end": match.end(),
0131:         })
0132:     return sorted(rates, key=lambda rate: rate["start"])

0139: def _named_rate_introductions(segment: str, rates: list[dict]) -> dict | None:
0140:     """只接受同一句由不同甲方/乙方标签分别引导的两个费率。"""
0141:     if re.search(r"丙方|丁方|第三方", segment):
0142:         return None

0145:         clause_start = max(
0146:             segment.rfind(char, 0, rate["start"]) for char in "，,；;。！？!?：:"
0147:         ) + 1

0158: def _named_party_rate_comparison(segment: str, rates: list[dict]):
0159:     """只在同一句两次具名引入后，由当事人标签解析一次明确的比率倍数比较。"""
0160:     introduced = _named_rate_introductions(segment, rates)
0161:     if introduced is None:
0162:         return None
0163:     comparisons = list(_NAMED_RATE_COMPARISON_RE.finditer(segment, rates[1]["end"]))
0164:     if len(comparisons) != 1:
0165:         return None
0166:     comparison = comparisons[0]
0167:     left_party = comparison.group("left_party")
0168:     right_party = comparison.group("right_party")
0169:     if left_party == right_party or set(introduced) != {left_party, right_party}:
0170:         return None
0171:     claimed = _exact_number(comparison.group("multiple"), allow_zero=False)
0172:     if claimed is None:
0173:         return None
0174:     actual = introduced[left_party]["value"] / introduced[right_party]["value"]
0175:     return comparison.start(), comparison.end(), claimed, actual, f"{left_party}/{right_party}"

0178: def _two_named_rate_difference(segment: str, rates: list[dict]):
0179:     """识别“两者相差N倍”，只在N与四种可支持口径均不一致时返回拦截依据。"""
0180:     if (len(rates) != 2 or any(rate["value"] == 0 for rate in rates)
0181:             or rates[0]["unit"] is None or rates[0]["unit"] != rates[1]["unit"]
0182:             or len(_RATE_MARKER_RE.findall(segment)) != 2):
0183:         return None
0184:     if _named_rate_introductions(segment, rates) is None:
0185:         return None
0186:     comparisons = list(_TWO_RATE_DIFFERENCE_RE.finditer(segment))
0187:     if len(comparisons) != 1 or comparisons[0].start() < rates[1]["end"]:
0188:         return None
0189:     comparison = comparisons[0]
0190:     claimed = _exact_number(comparison.group("multiple"), allow_zero=False)
0191:     if claimed is None:
0192:         return None
0193:     former, latter = rates[0]["value"], rates[1]["value"]
0194:     gap = abs(former - latter)
0195:     interpretations = (
0196:         ("前者/后者", former / latter),
0197:         ("后者/前者", latter / former),
0198:         ("差额/前者", gap / former),
0199:         ("差额/后者", gap / latter),
0200:     )
0201:     if any(value == claimed for _, value in interpretations):
0202:         return None
0203:     return comparison.start(), comparison.end(), claimed, interpretations

0424: def _save_locked(tasks, root: str, case_id: str, task: dict, title: str, content: str,
0425:                  *, check_fn, coverage_fn) -> dict:
0426:     """调用方持有来源任务锁；只写一个新版本，并在 metadata 失败时收回本次新文件。"""
0427:     if task.get("skill") == "contract-review":
0428:         mismatch = _rate_ratio_mismatch(content)
0429:         if mismatch is not None:
0430:             raise ApiError("INVALID_ARGUMENT", "rate_ratio_mismatch", mismatch)
0431:     tid = task["task_id"]
0432:     drafts_rel = tasks.rel(tid, "草稿")
0433:     version = tasks.next_draft_version(root, title, ".md")   # 本案同标题接着往上加，不分任务
0434:     rel = f"{drafts_rel}/{title}-v{version}.md"
0435:     target = gate.resolve_write(root, rel, op="save_draft")
0436:     if target.exists():
0437:         raise ApiError("INVALID_ARGUMENT", "target_exists")
0438:     gate.write_bytes(root, rel, content.encode("utf-8"), op="save_draft")  # 标题是设备名等：闸门拒绝
0439:     try:
0440:         check, cites = check_fn()
0441:         cov = coverage_fn()
0442:         res = tasks.result(root, tid)
0443:         res["drafts"].append({"title": title, "path": rel, "version": version})
0444:         res.update(citation_check=check, coverage=cov, citations=merge_citations(res["citations"], cites))
0445:         tasks.save_result(root, res)
0446:     except BaseException:
0447:         try:
0448:             gate.delete_work_file(root, rel, op="save_draft_rollback")
0449:         except Exception as exc:  # noqa: BLE001 保留原异常；清理失败只记类型
0450:             logs.event("draft", "rollback", status="fail", case_id=case_id, error=type(exc).__name__)
0451:         raise
0452:     # 正式草稿存好了：进度文件清掉（T14 派修；之后模型再回复，progress 会重写，task/end 时再清）
0453:     gate.delete_work_file(root, f"{drafts_rel}/{IN_PROGRESS}", op="save_draft")
0454:     not_fully = [p["name"] for p in cov["partially_read"]] + cov["not_read"]
0455:     return {"path": rel, "version": version, "citation_check": check, "coverage": cov, "not_fully_read": not_fully}
