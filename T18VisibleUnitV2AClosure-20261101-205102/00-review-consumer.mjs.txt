0507 async function verifyPinnedSources() {
0508   let manifest;
0509   try {
0510     manifest = JSON.parse(await fs.readFile(new URL(MANIFEST_NAME, import.meta.url), "utf8"));
0511   } catch {
0512     fail("source", "SOURCE_MANIFEST_INVALID");
0513   }
0514   if (!manifest || manifest.algorithm !== "sha256") fail("source", "SOURCE_MANIFEST_INVALID");
0515   const pins = [
0516     ["consumer", CONSUMER_NAME, manifest.consumer],
0517     ["fixed_data", FIXED_DATA_NAME, manifest.fixed_data],
0518     ["offline_controls", CONTROLS_NAME, manifest.offline_controls],
0519   ];
0520   let fixedDataBytes;
0521   for (const [label, fileName, pin] of pins) {
0522     if (!pin || pin.path !== fileName || !Number.isSafeInteger(pin.bytes) ||
0523         pin.bytes < 0 || !/^[0-9a-f]{64}$/.test(pin.sha256)) {
0524       fail("source", "SOURCE_MANIFEST_INVALID");
0525     }
0526     const bytes = await fs.readFile(new URL(fileName, import.meta.url));
0527     const actual = { bytes: bytes.byteLength, sha256: crypto.createHash("sha256").update(bytes).digest("hex") };
0528     if (actual.bytes !== pin.bytes || actual.sha256 !== pin.sha256) fail("source", `${label.toUpperCase()}_PIN_MISMATCH`);
0529     if (label === "fixed_data") fixedDataBytes = bytes;
0530   }
0531   return fixedDataBytes.toString("utf8");
0532 }

0572 async function readCredentialFromStdin() {
0573   if (process.stdin.isTTY) fail("credential", "TTY_NOT_ALLOWED");
0574   return new Promise((resolve, reject) => {
0575     let captured = Buffer.alloc(0);
0576     let settled = false;
0577     const cleanup = () => {
0578       process.stdin.off("data", onData);
0579       process.stdin.off("end", onEnd);
0580       process.stdin.off("error", onError);
0581       if (typeof process.stdin.pause === "function") process.stdin.pause();
0582     };
0583     const finish = () => {
0584       if (settled) return;
0585       settled = true;
0586       cleanup();
0587       let line = captured;
0588       if (line.length > 0 && line[line.length - 1] === 13) line = line.subarray(0, -1);
0589       if (line.length === 0 || line.length > 4096 ||
0590           [...line].some((byte) => byte < 33 || byte > 126)) {
0591         reject(new PilotProtocolError("credential", "CREDENTIAL_INVALID"));
0592         return;
0593       }
0594       resolve(line.toString("ascii"));
0595     };
0596     const onData = (chunk) => {
0597       if (settled) return;
0598       const bytes = Buffer.from(chunk);
0599       captured = Buffer.concat([captured, bytes]);
0600       const newline = captured.indexOf(10);
0601       if (newline >= 0) {
0602         captured = captured.subarray(0, newline);
0603         finish();
0604       } else if (captured.length > 4096) {
0605         settled = true;
0606         cleanup();
0607         reject(new PilotProtocolError("credential", "CREDENTIAL_INVALID"));
0608       }
0609     };
0610     const onEnd = () => finish();
0611     const onError = () => {
0612       if (settled) return;
0613       settled = true;
0614       cleanup();
0615       reject(new PilotProtocolError("credential", "STDIN_FAILED"));
0616     };
0617     process.stdin.on("data", onData);
0618     process.stdin.on("end", onEnd);
0619     process.stdin.on("error", onError);
0620     process.stdin.resume();
0621   });
0622 }

0624 export async function performRequest(stage, requestBody, credential, deadline) {
0625   const startedAt = Date.now();
0626   const timeoutMs = requestTimeoutForDeadline(startedAt, deadline);
0627   if (timeoutMs <= 0) fail(stage, "ENTIRE_CALL_CAP_EXCEEDED");
0628   const signal = AbortSignal.timeout(timeoutMs);
0629   let responseStatus = null;
0630   const chunks = [];
0631   let capturedBytes = 0;
0632   let overflow = false;
0633   try {
0634     const response = await fetch(ROUTE, {
0635       method: "POST",
0636       redirect: "error",
0637       headers: { Authorization: buildAuthorizationHeader(credential), "Content-Type": "application/json" },
0638       body: requestBody,
0639       signal,
0640     });
0641     responseStatus = response.status;
0642     if (response.body) {
0643       const reader = response.body.getReader();
0644       let streamFailed = false;
0645       try {
0646         while (true) {
0647           const { done, value } = await reader.read();
0648           if (done) break;
0649           const chunk = Buffer.from(value);
0650           if (capturedBytes + chunk.byteLength > RESPONSE_CAP_BYTES) {
0651             overflow = true;
0652             await reader.cancel();
0653             break;
0654           }
0655           chunks.push(chunk);
0656           capturedBytes += chunk.byteLength;
0657         }
0658       } catch (error) {
0659         streamFailed = true;
0660         throw error;
0661       } finally {
0662         if (streamFailed) {
0663           try {
0664             await reader.cancel();
0665           } catch {
0666             // Cancel errors are swallowed only for a stream that already failed.
0667           }
0668         }
0669         reader.releaseLock();
0670       }
0671     }
0672     return {
0673       status: response.status,
0674       body: Buffer.concat(chunks).toString("utf8"),
0675       capturedBytes,
0676       responseOverflow: overflow,
0677       durationMs: Date.now() - startedAt,
0678       transportError: null,
0679     };
0680   } catch (error) {
0681     const signalTimedOut = signal.aborted === true &&
0682       signal.reason && signal.reason.name === "TimeoutError";
0683     const timedOut = (error && error.name === "TimeoutError") || signalTimedOut;
0684     const wholeDeadlineExpired = Date.now() >= deadline ||
0685       (timedOut && timeoutMs < PER_CALL_TIMEOUT_MS);
0686     return {
0687       status: responseStatus,
0688       body: Buffer.concat(chunks).toString("utf8"),
0689       capturedBytes,
0690       responseOverflow: overflow,
0691       durationMs: Date.now() - startedAt,
0692       transportError: wholeDeadlineExpired
0693         ? "ENTIRE_CALL_CAP_EXCEEDED"
0694         : timedOut ? "TIMEOUT" : "TRANSPORT_FAILURE",
0695     };
0696   }
0697 }

0860 function isMainModule() {
0861   if (!process.argv[1]) return false;
0862   const moduleUrl = new URL(import.meta.url);
0863   if (moduleUrl.protocol !== "file:") return false;

0865   if (process.platform === "win32" && /^\/[A-Za-z]:\//.test(modulePath)) modulePath = modulePath.slice(1);
0866   const invoked = path.resolve(process.argv[1]);
0867   const current = path.normalize(modulePath);
0868   return process.platform === "win32"
0869     ? invoked.toLowerCase() === current.toLowerCase()
0870     : invoked === current;
